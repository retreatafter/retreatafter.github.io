1.资源导入

  • (1)这里所说的资源大多为美术资源，若导入的是完整资料，通常还包括了代码、音效等。
  • (2)在线导入Unity资源
    • 1)编辑器中“窗口Window-资源商店AssetStore”位置跳转至资源网址。
    • 2)直接进入 Unity资源商店
    • 3)回到编辑器，这时可以在"窗口Window-包管理器MyAssets"种看到所拥有的所有包资源，进行下载。
    • 4)下载完成后可以看到导入的包资源的具体内容，一步步确认即可。
  • (3)导入本地Unity资源
    • 1)直接将[.unitypackage]包拖拽至Assets位置。
  • (4)细节
    • 项目导入的资源中如果已经包括了光源的，记得将默认自带的光源删除，否则会产生过曝等问题。光源调整位于窗口Window-渲染Rendering-光源Lighting下的环境照明源EnvironmentLightingSource。本次将其按需修改，从天空盒改为源为自定义颜色。

// 实现子弹的自行销毁
public class Destroy : MonoBehaviour
{
    public float topBound = 20; //上边界
    public float lowerBound = -20; //下边界

    void Start()
    {
        
    }

    void Update()
    {
        // 当按下Space键时执行循环中的操作，GetKeyDown()方法是检测一次，GetKey()方法是只要按键不抬起就持续触发检测
        if (transform.position.z > topBound)
        {
            Destroy(gameObject);
        }else if(transform.position.z < lowerBound){
            Destroy(gameObject);
        }
    }

}

public class aiEnemy : MonoBehaviour
{
		public GameObject[] animalPrefabs; //2.数组用于存放不同动物，具体的存放哪些可以在Unity界面进行拖动选择
		public float xRange = 10; //3.范围
    
    void Start()
    {
        InvokeRepeating("SpawnRandomAnimal",2,3); //InvokeRepeating()方法可以反复调用某个事件，参数分别是：字符String的方法名，首次调用的延迟时间（例如这里的2即游戏开始后2秒开始首次调用此方法），后续调用的间隔时间，也可以将参数抽取为一个可以调整的公共变量
    }

    void Update()
    {

    }
		
		void SpawnRandomAnimal()
		{
				int animalIndex = Random.Range(0,animalPrefabs.Length); //4.动物的种类，并进行随机抽取
				float randomX = Random.Range(-xRange,xRange); //5.动物位置，并进行随机生成
				Instantiate(animalPrefabs[animalIndex],newVector3(randomX,0,20),animalPrefabs[animalIndex].transform.rotation); //1.生成物体的函数Instantiate，参数分别是：生成的物体，生成范围，旋转角度
		}
}

public class PlayerController : MonoBehaviour
{
		public float jumpForce = 10;
		public float gravityModifer = 2;
		private Rigidbody rgd;
		private bool isGround = true; //判断角色是否在地面，限制其在空中的多段跳
		
    void Start()
    {
				rgd = GetComponent<Rigidbody>(); //获取Rigidbody组件
				Physics.gravity *= gravityModifer; //调整重力加速度
    }

    void Update()
    {
				if(Input.GetKeyDown(KeyCode.Space) && isGround)
				{
						rgd.AddForce(Vector3.up * jumpForce, ForceMode.Impulse); // ForceMode.Impulse是施加力的模式，此处为瞬时施加的冲击力，用来模拟跳跃时的情况
						isGround= false; //跳跃时是肯定不在地面上的
				}
    }
    
		private void OnCollisionEnter(Collision collision)
		{
				isGround = true; //一次简单的碰撞检测，来证明物体在地面上
		}
}

public class RepeatBackground : MonoBehaviour
{
		private Vector3 startPosition;
		private float xBound;
		
    void Start()
    {
        start Position = transform.position;
				xBound = startPosition.x - GetComponent<BoxCollider>().size.x/2; //这里去获取BoxCollider组件是因为可以获得背景图片的精确坐标，也就能精确判断中心点的位置
    }

    void Update()
    {
		    if(transform.position.x < xBound)
				{
						transform.position = startPosition;
				}
    }
}

public class Controller : MonoBehaviour
{
		private Rigidbody rgd;
		private Gameobject player;
		public float moveForce = 10;
		
		void Start()
		{
				rgd = GetComponent<Rigidbody>();
				player = GameObject.Find("Player”); // 通过Find()方法来查找游戏物体Player，这里通过代码赋值，和在inspect界面下拖拽赋值效果一致，且更推荐代码赋值
		}
		
    void Update()
    {
				Vector3 LookDirection = (player.transform.position - transform.position).normalized; //确认物体追踪的方向并归一化向量，确认施加力的方向，之后再确定具体施加的力的大小
				rgd.AddForce(lookDirection * moveForce); //moveForce即确定具体施加的力的大小
    }
}

public class Controller : MonoBehaviour
{
				//部分前置代码省略
				private GameObject focalPoint;
				public bool hasPowerup = false; //用于判断奖励条件是否开启
				public float powerupStrength = 15;
		
		void Start()
		{
				//部分前置代码省略
				focalPoint = GameObject.Find("FocalPoint”);
		}
		
    void Update()
    {
				Vector3 LookDirection = (player.transform.position - transform.position).normalized; //确认物体追踪的方向并归一化向量，确认施加力的方向，之后再确定具体施加的力的大小
				rgd.AddForce(lookDirection * moveForce); //moveForce即确定具体施加的力的大小
    }
}

		private void OnTriggerEnter(Collider other)
		{
				if(other.CompareTag("Powerup”))
				{
						Destroy(other.gameObject);
						hasPowerup = true;
						StartCoroutine(PowerupCountDown()); //执行PowerupCountDown()方法，这里为固定写法
				}
		}

		private void OnCollisionEnter(Collision collision)
		{
				if(collision.gameObject.CompareTag("Enemy”) && hasPowerup)
				{
						Rigidbody enemyRgd = collision.gameObject.GetComponent<Rigidbody>();
						Vector3 enemyDirection = (collision.gameObject.transform.position - transform.position).normalized;
						enemyRgd.AddForce(enemyDirection * powerupStrength ,ForceMode.Impulse);
				}
		}

		IEnumerator PowerupCountDown()
		{
				yield return new WaitForSeconds(7); //7秒后执行下述代码，除参数外，这里为固定写法
				hasPowerup = false;
		}

public class Controller : MonoBehaviour
{
		public GameObject enemyPrefab;
		public float range = 5;
		public int spawnCount = 1;
		
		void Start()
		{
				SpawnRandomEnemy(spawnCount); //生成敌人
		}
		
		private void Update()
		{
				int enemyCount = FindObjectsByType<EnemyController>(FindObjectsSortMode.None).Length; //获取当前敌人数量，用数组储存，FindObjectsSortMode.None是查找的方式
				//注意，敌人销毁的代码已省略
				if(enemyCount == 0)
				{
						spawnCount++;
						SpawnRandomEnemy(spawnCount);
				}
		}
		
		//敌人生成
		void SpawnRandomEnemy(int number)
		{
				for(int i =0; i < number; i++)
				{
						float randomX = Random.Range(-range,range);
						float randomZ = Random.Range(-range,range);
						Vector3 postion = newVector3(randomX,0,randomZ);
						Instantiate(enemyPrefab,postion,enemyPrefab.transform.rotation);
				}
		}
}
