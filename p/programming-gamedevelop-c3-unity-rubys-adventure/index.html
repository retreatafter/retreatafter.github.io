public class RubyController: MonoBehaviour
{
		//基础数据
		private Rigidbody2D rigidbody2d;
		public float speed;
		public int maxHealth = 5;
		private int currentHealth;
		
		public int Health { get { return currentHealth; } }
		
		//Ruby的无敌时间相关数据
		public float timeInvincible = 2.0f; //定义无敌时间，时间常量，但这里的2其实是指数学意义上的2，而非时间意义上的2，因此接下来需要设置一个“计时器”
		public bool isInvincible; //布尔类型，“开关”，方便我们调整启动无敌时间的时机
		public float invincibleTimer; //计时器，变量，我们以下用数值“模拟”时间的概念来实现计时器效果
		
		//出生点数据
		private Vector3 respawnPosition;
		
		void Start()
		{
				rigidbody2d = GetComponent<Rigidbody2D>();
				currentHealth = maxHealth;
				respawnPosition = transform.position; //初始位置
		}
		
		void Update()
		{
				// 键盘控制移动
				float horizontal = Input.GetAxis("Horizontal");
				float vertical = Input.GetAxis("Vertical");
				//移动
				Vector2 position = transform.position;
				position.x = position.x + speed*horizontal*Time.deltaTime;
				position.y = position.y + speed*vertical*Time.deltaTime;
				// transform.position = position;
				rigidbody2d.MovePosition(position);
				
				//无敌时间计算
				if (isInvincible) //进入无敌时间
				{
						invincibleTimer = invincibleTimer - Time.deltaTime; //Time.deltaTime可视为每帧增量，本质也是一个值，约为1/60，那么Update()每秒执行约60次，Time.deltaTime也就模拟出了“秒”的概念，此处表示invincibleTimer无敌时间每帧递减
						if (invincibleTimer<=0) //无敌时间结束
						{
								isInvincible = false;
						}
				}
		}
		
		//血量变动
		public void ChangeHealth(int amount)
		{
				if (amount<0) // 受到伤害
				{
						if (isInvincible) //正处于无敌时间时
						{
								return; //这里return后下方代码不会再运行
						}
		
						isInvincible = true; //开启无敌状态
						invincibleTimer = timeInvincible; //开始计时
				}
				currentHealth = Mathf.Clamp(currentHealth+amount,0,maxHealth);
        if (currentHealth<=0) //血量归零时
        {
            Respawn(); //调用复活方法
        }
		}
		
		//复活
    private void Respawn()
    {
        ChangeHealth(maxHealth); //血量恢复满格
        transform.position = respawnPosition; //回到初始位置
    }
}

public class EnemyController : MonoBehaviour
{
    //基础数据
    public float speed=3; //移动速度
    private Rigidbody2D rigidbody2d;
    public bool vertical; //轴向控制
    private int direction = 1; //方向控制，正数表示正方向
    public float changeTime = 3.0f; //方向改变的时间间隔，常量
    private float timer; //计时器
    
	  void Start()
		{
				rigidbody2d = GetComponent<Rigidbody2D>(); //获取机器人身上的Rigidbody2D组件
				timer = changeTime; //游戏刚开始的初始值
		}
		
		void Update()
		{
	      //方向自动变向
	      timer -= Time.deltaTime; //倒计时，控制方向改变
	      
	      if (timer<0) //倒计时结束
        {
            direction = -direction; //变向
            timer = changeTime; //更改方向后重新计时
        }
        
	      Vector2 position = rigidbody2d.position; //物体的当前位置
	      
        if (vertical) //垂直轴向
        {
            position.y = position.y + Time.deltaTime * speed*direction;
        }
        else //水平轴向
        {
            position.x = position.x + Time.deltaTime * speed*direction;
        }
    
        rigidbody2d.MovePosition(position); //改变物体位置
		}
}

				// position.y = position.y + speed*vertical*Time.deltaTime;
