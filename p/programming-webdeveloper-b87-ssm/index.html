1.概述

  • (1)SSM整合：Springmvc框架对应视图层控制Controller，Spring框架对应Service业务层，Mybatis框架对应dao持久层。视图层Controller接收用户请求，调用Service处理业务逻辑，Service调用dao层访问数据库并返回。

2.步骤分析

  • 1)搭建环境。
    • （持久层）SQL数据库MySQL建数据库，创建表，添加数据，并配置IDEA中数据库连接。
    • 创建maven的web项目。
    • pom.xml文件导入依赖：Springmvc依赖、Spring依赖、mybatis依赖、Jackson依赖、mysql驱动、druid连接池依赖、jsp依赖、servlet依赖。
  • 2)配置文件。
    • web.xml文件中注册：
      • 中央调度器（前端控制器）对象DispatcherServlet。→创建springmvc容器对象，后续才能创建controller对象；创建servlet，接收用户请求。
      • spring的监听器对象ContextLoaderListener。→创建spring容器对象，后续才能创建service和dao等对象。
      • 字符集过滤器。→解决post请求乱码问题。
    • 编写Springmvc主配置文件：
      • 声明组件扫描器，指定@Controller所在包名位置。
      • 声明视图解析器，指定简化路径的书写。
      • 声明注解驱动，以支持ajax请求、处理静态资源。
    • 编写Spring主配置文件：
      • 声明数据源DataSource，连接数据库。
      • 声明mybatis中SqlSessionFactoryBean类。
      • 声明组件扫描器。
      • 声明service所在位置。
    • 编写Mybatis主配置文件：
      • 声明扫描mapper映射文件的位置。
      • 其它可选的辅助功能。
  • 3)测试代码业务实现。
    • 基本三层结构项目包及对应的接口和实现类：
      • 定义pojo实体类，对应数据库表Student，一个对象就是一行数据，一张表对应一个实体类。
      • 创建持久层的dao（mapper）接口，定义操作数据库的方法，及对应的sql映射文件mapper.xml。
      • 创建Service接口和实现类，业务层调用dao接口。
      • 创建控制器类Controller。控制器类Controller使用到@Controller注解声明为Bean由Spring创建管理对象和@RequestMapping注解绑定请求地址和方法。
    • 创建发起请求的页面jsp和作为显示请求处理结果的jsp。

//applicationContext.xml配置文件
<?xml version="1.0" encoding="UTF-8"?>
<beans xmlns="http://www.springframework.org/schema/beans"
       xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
       xmlns:context="http://www.springframework.org/schema/context"
       xsi:schemaLocation="http://www.springframework.org/schema/beans http://www.springframework.org/schema/beans/spring-beans.xsd">

    <!--声明数据源DataSource，连接数据库，格式为固定写法-->
    <context:property-placeholder location="classpath:conf/jdbc.properties"/>
		<bean id="dataSource" class="com.alibaba.druid.pool.DruidDataSource" init-method="init" destroy-method="close">
		        <!--以下基础信息拆分为单独的配置文件-->
		        <property name="url" value="${jdbc.url}"/>
		        <property name="username" value="${jdbc.username}"/>
		        <property name="password" value="${jdbc.password}"/>
		</bean>

    <!--声明mybatis中SqlSessionFactoryBean类，用于创建SqlSessionFactory，格式为固定写法-->
    <bean id="SqlSessionFactory" class="org.mybatis.spring.SqlSessionFactoryBean">
            <!--set注入，把DruidDataSource赋给dataSource属性，得到数据库基本信息-->
		        <property name="dataSource" ref="dataSource"/>
		        <!--mybatis主配置文件位置-->
		        <property name="configLocation" value="classpath:mybatis.xml"/>
    </bean>
    
    <!--声明扫描器，创建dao对象，生成每个dao接口的代理对象，格式为固定写法-->
    <bean class="org.mybatis.spring.mapper.MapperScannerConfigurer">
            <!--指定SqlSessionFactory对象的id-->
		        <property name="SqlSessionFactoryBeanName" value="sqlSessionFactory"/>
		        <!--dao包位置，MapperScannerConfigurer将会扫描包中所有接口并执行一次getMapper()方法，得到dao对象-->
		        <property name="basePackage" value="com.retreatafter.dao"/>
    </bean>

    <!--声明service所在位置-->
    <bean context:component-scan base-package="com.retreatafter.service" />

</beans>

//定义表现层控制器bean
@Controller
@RequestMapping("/student")
public class StudentController {
    
    //引用类型自动注入
    @Autowired
    private StudentService service;
    
    //处理注册业务
    //请求映射注解@RequestMapping，绑定请求地址和方法，表示该地址传来的请求由该方法处理
    @RequestMapping("/addStudent.do")
    public ModelAndView addStudent(Student student){
        ModelAndView mv = new ModelAndView(); //视图对象
        String tips = "注册失败"; //提示
        //调用service
        int nums = service.addStudent(student);
        if(nums > 0){
            //注册成功，更换提示
            tips = "学生【" + student.getName +"】注册成功";
        }
        //添加数据
        mv.addObject("tips",tips);
        //指定结果页面
        mv.setViewName("result");
        //返回
        return mv;
    }
    
    //处理查询业务，响应ajax
    @RequestMapping("/queryStudent.do")
    @ResponseBody
    public List<Student> queryStudent(){
        //参数检查，进行简单的数据处理
        List<Student> students = service.findStudents();
        return students;
    }
}
