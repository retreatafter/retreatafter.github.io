using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;

public class Move : MonoBehaviour
{
    public Rigidbody rb = null; 
    public float moveSpeed = 5;
    public GameObject vfx;
    public int score = 0;
    public Text scoreText;
    public GameObject winText;
    

    // Start is called once before the first execution of Update after the MonoBehaviour is created
    void Start()
    {
        rb = GetComponent<Rigidbody>();
    }

    // Update is called once per frame
    void Update()
    {
        float h = Input.GetAxis("Horizontal");
        float v = Input.GetAxis("Vertical");
        rb.AddForce(new Vector3(h, 0, v) * moveSpeed);
    }

    private void OnTriggerEnter(Collider other)
    {
        if (other.tag == "Food")
        {
            Destroy(other.gameObject);
            Instantiate(vfx,transform.position,Quaternion.identity);
            score++;
            scoreText.text = "score:" + score;
            if (score == 10)
            {
                winText.SetActive(true);
            }
        }

    }
}

public class Move: MonoBehaviour
{
	public Rigidbody rb = null; //定义一个刚体类型Rigidbody的物品对象，且访问修饰符设置为public，这样这个变量即可在Unity的Inspect中显示
	
	void Start()
	{
		rb = GetComponent<Rigidbody>(); //得到组件方法GetComponent<>()，需要关联的是刚体组件Rigidbody，故GetComponent<Rigidbody>()再将其赋值给rb，这样就关联起来了rb和刚体组件
	}
	
	void Update()
	{
		rb.AddForce(Vector3.right); 
		//可以看到AddForce()是Rigidbody类中自带的方法，rb已经有了刚体组件的效果，在这里给对象rb添加力就能使其运动起来
		//进一步的，给出力的方向和大小，力有大小和方向，向量也有，因此这里用向量方法演示施加力。Vector3.right相当于默认给了一个三维向量固定向右1牛的力，且放在Update()方法中相当于给一个持续施加的力
		//类似的，new Vector3(2,0,0)相当于new了一个Vector3对象，给了一个x轴为2牛的力。
	}
}

public class Move: MonoBehaviour
{
	public Rigidbody rb = null; //定义一个刚体类型Rigidbody的物品对象，且访问修饰符设置为public，这样这个变量即可在Unity的Inspect中显示
	public float moveSpeed = 10; //设定了物品对象的初速度，也可以不设定，不设定就会导致物体速度无法调整
	
	void Start()
	{
		rb = GetComponent<Rigidbody>(); //得到组件方法GetComponent<>()，需要关联的是刚体组件Rigidbody，故GetComponent<Rigidbody>()再将其赋值给rb，这样就关联起来了rb和刚体组件
	}
	
	void Update()
	{
		float h = Input.GetAxis("Horizontal");
		//键盘接收获取水平左右输入轴的数值，默认为键盘的[A][D]/[←][→]键。GetAxis()方法为获取轴，这个方法属于Input类下，获得的值是一个浮点数float，将其赋值给h。这个值的范围为[-1,1]
		float v = Input.GetAxis("Vertical");
		//键盘接收获取垂直前后输入轴的数值，默认为键盘的[W][S]/[↑][↓]键。
		rb.AddForce(new Vector3(h,0,v) * moveSpeed); 
		//将监听键盘的结果转换为施加给物体对象的力
		//但通过这种方式直接转换的力，由于力绝对值的范围最高是1，因此我们想让物体对象运动速度更快，可以让这个向量值做乘法
	}
}

3.相机跟随

  • (1)原理为向量加法。算出相机与物体的向量偏移，再用物体位置加上偏移位置，将这个位置赋值给相机，即能得到相机位置。

public class FollowTarget : MonoBehaviour
{
    public Transform pt; //设定一个变量用于接收到要跟随的物体位置，具体要跟随哪一个物体位置可以在Unity检查器Inspect直接修改
    private Vector3 offset; //声明偏移量，在这个地方先定义，在不同的方法中才能使用，同时，这个变量我们不需要修改，因此访问修饰符为私有private
    
    void Start()
    {
        offset = transform.position - pt.position; //设置一个三维向量Vector3类的变量offset，transform.position来表示相机位置，减去要跟随的物体位置即为偏移量，算出这个初始偏移量
    }

    
    void Update()
    {
        transform.position = offset + pt.position; //物体不断地运动，相机做出跟随
    }
}
  • (3)邪道：也可以让相机物体成为游戏物体的子物体，让二者成为一体，也可以实现相机跟随。

public class Bound : MonoBehaviour
{
    public float speed = 10;
    public float xRange = 10;

    void Start()
    {
        
    }


    void Update()
    {
		    // 一旦游戏物体超过边界值时就让物体回到边界
        if (transform.position.x < -xRange)
        {
            transform.position = new Vector3(-xRange, transform.position.y, transform.position.z);
            // 在有些情况中需要速度归零，playerRB.linearVelocity = Vector3.zero; 其中playerRB是Rigidbody类的对象
        }
        if (transform.position.x > -xRange)
        {
            transform.position = new Vector3(xRange, transform.position.y, transform.position.z);
        }
    }
}

    • 触发器本质也是一种碰撞体，开启触发器处于碰撞组件Collider内，即一个物体可以由碰撞盒切换为触发器。
    • 同样，Unity提供的触发器的系统事件类还有OnTriggerExit类，可以检测触发已离开；OnTriggerStay类，可以检测接触的持续时。更详细的可以阅读官方手册。
